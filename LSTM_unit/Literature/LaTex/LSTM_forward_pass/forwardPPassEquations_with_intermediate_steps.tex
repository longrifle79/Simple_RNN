\subsection*{The Six Equations Used at Every Time Step}

\begin{align*}
f_t &= \sigma(w_{hf} \cdot h_{t-1} + w_{xf} \cdot x_t + b_f) \\
i_t &= \sigma(w_{hi1} \cdot h_{t-1} + w_{xi1} \cdot x_t + b_{i1}) \\
\tilde{C}_t &= \tanh(w_{hi2} \cdot h_{t-1} + w_{xi2} \cdot x_t + b_{i2}) \\
o_t &= \sigma(w_{ho} \cdot h_{t-1} + w_{xo} \cdot x_t + b_o) \\
C_t &= f_t \odot C_{t-1} + i_t \odot \tilde{C}_t \\
h_t &= o_t \odot \tanh(C_t)
\end{align*}

(where $\sigma(z) = \frac{1}{1 + e^{-z}}$ and $\tanh$ is the standard hyperbolic tangent function).

\subsection*{Full Forward Pass with Every Intermediate Step}

\subsubsection*{Time Step $t = 1$}

\textbf{Inputs:} $h_0 = 0$, $C_0 = 0$, $x_1 = 0.5$

\textcolor{headingRed}{\textbf{Forget gate}}
\begin{itemize}[leftmargin=*]
    \item Linear combination: $0.28 \times 0 + (-0.15) \times 0.5 + 0.05 = -0.075 + 0.05 = -0.025$
    \item $f_1 = \sigma(-0.025) = \frac{1}{1 + e^{0.025}} \approx \frac{1}{2.025315} \approx 0.4938$
\end{itemize}

\textcolor{headingRed}{\textbf{Input gate (sigmoid)}}
\begin{itemize}[leftmargin=*]
    \item Linear combination: $0.35 \times 0 + 0.42 \times 0.5 + 0.18 = 0.21 + 0.18 = 0.39$
    \item $i_1 = \sigma(0.39) \approx 0.5963$
\end{itemize}

\textcolor{headingRed}{\textbf{Candidate}}
\begin{itemize}[leftmargin=*]
    \item Linear combination: $0.22 \times 0 + 0.31 \times 0.5 + 0.09 = 0.155 + 0.09 = 0.245$
    \item $\tilde{C}_1 = \tanh(0.245) \approx 0.2404$
\end{itemize}

\textcolor{headingRed}{\textbf{Output gate}}
\begin{itemize}[leftmargin=*]
    \item Linear combination: $0.47 \times 0 + 0.19 \times 0.5 + 0.26 = 0.095 + 0.26 = 0.355$
    \item $o_1 = \sigma(0.355) \approx 0.5878$
\end{itemize}

\textcolor{headingRed}{\textbf{Cell state update}}
\begin{itemize}[leftmargin=*]
    \item $C_1 = (0.4938 \times 0) + (0.5963 \times 0.2404) = 0 + 0.1434 = 0.1434$
\end{itemize}

\textcolor{headingRed}{\textbf{Hidden state update}}
\begin{itemize}[leftmargin=*]
    \item $h_1 = 0.5878 \times \tanh(0.1434) \approx 0.5878 \times 0.1426 = 0.0838$
\end{itemize}

\textbf{Updated after $t=1$:} $h_1 \approx 0.0838$, $C_1 \approx 0.1434$
